async def backup_existing() -> None:

    backup_dir = f"/data/ax/ax_novas/data/backups/{timestamp}"
    
    if Path("/data/ax/ax_novas/data").exists():
        console.print("Creating backup of existing data...")
        Path(backup_dir).mkdir(parents=True, exist_ok=True)
        os.system(f"cp -r /data/ax/ax_novas/data/* {backup_dir}/")
        console.print(f"Backup created at: {backup_dir}")


async def verify_environment() -> bool:
    """Verify environment variables."""
    required_vars = [
        "OPENAI_API_KEY",
        "ANTHROPIC_API_KEY",
        "NOVA_ENVIRONMENT",
        "NOVA_LOG_LEVEL"
    ]
    
    missing = [var for var in required_vars if not os.getenv(var)]
    
    if missing:
        console.print(
            "[bold red]Missing environment variables:[/bold red]"
        )
        for var in missing:
            console.print(f"- {var}")
        return False
        
    return True

async def verify_system_resources() -> bool:
    """Verify system resources are adequate."""
    cpu_count = psutil.cpu_count()
    memory = psutil.virtual_memory()
    
    console.print(f"[bold blue]System Resources:[/bold blue]")
    console.print(f"CPUs: {cpu_count}")
    console.print(f"Memory: {memory.total / (1024**3):.1f}GB")

    # Check if we have enough resources
    required_memory = sum(
        int(res["memory_limit"].replace("G", ""))
        for res in AGENT_RESOURCES.values()
    )
    
    if memory.available / (1024**3) < required_memory:
        console.print(
            f"[bold red]Insufficient memory. Need {required_memory}GB, "
            f"have {memory.available / (1024**3):.1f}GB available[/bold red]"
        )
        return False
        
    if cpu_count < 88:
        console.print(

        return False
        
    return True

async def verify_agent_health(
    agent: Agent,
    monitoring_service: MonitoringService
) -> bool:
    """Verify agent is healthy after launch."""
    # Check agent status
    health = await monitoring_service.check_health()
    agent_health = health["components"].get(f"agent_{agent.name}")
    
    if not agent_health:
        return False
        
    return agent_health.status == HealthStatus.HEALTHY


async def configure_agent_resources(
    agent: Agent,
    resources: Dict[str, str]
) -> None:
    """Configure system resources for agent."""
    process = psutil.Process()
    
    # Set CPU priority
    cpu_priority = int(resources["cpu_priority"])
    os.nice(cpu_priority)
    
    # Set memory limit
    memory_limit = resources["memory_limit"]
    memory_bytes = int(memory_limit.replace("G", "")) * 1024 * 1024 * 1024
    
    import resource
    resource.setrlimit(
        resource.RLIMIT_AS,
        (memory_bytes, memory_bytes)
    )
    
    # Set CPU affinity to all cores
    process.cpu_affinity(list(range(psutil.cpu_count())))


async def launch_agent(
    app: NovaApp,
    agent_name: str,
    progress: Progress
) -> Optional[Agent]:
    """Launch a single agent with proper resource configuration."""
    task_id = progress.add_task(
        f"Launching {agent_name}...",
        total=None
    )
    
    try:
        # Get resource configuration
        resources = AGENT_RESOURCES[agent_name]
        
        # Create agent configuration
        config = AgentConfig(
            type=agent_name.split("_")[0],
            models=[resources["model"]],
            capabilities=[],  # Will be set based on type
            memory_config={
                "max_memories": 1000,
                "ttl": 86400,
                "consolidation_threshold": 100
            },
            rate_limits={
                "requests_per_minute": 60,
                "tokens_per_minute": 100000
            }
        )
        
        # Create agent
        agent = await app.create_agent(
            name=agent_name,
            config=config
        )
        
        # Configure system resources
        await configure_agent_resources(agent, resources)
        
        # Verify agent health
        if not await verify_agent_health(agent, app.monitoring_service):
            raise Exception(f"Agent {agent_name} failed health check")
            
        progress.update(
            task_id,
            description=f"[green]Launched {agent_name}[/green]"
        )
        return agent
        
    except Exception as e:
        progress.update(
            task_id,
            description=f"[red]Failed to launch {agent_name}: {str(e)}[/red]"
        )
        return None
