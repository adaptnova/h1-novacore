<content>"""Setup script for NOVA's tiered vector store system."""

import asyncio
import os
from pathlib import Path
from typing import Dict, List, Optional, Union

import faiss
import numpy as np
from chromadb import Client as ChromaClient
from chromadb.config import Settings as ChromaSettings
from pymilvus import (
    Collection,
    CollectionSchema,
    DataType,
    FieldSchema,
    connections,
    utility
)
from rich.console import Console
from rich.progress import Progress, SpinnerColumn, TextColumn

console = Console()

# Base paths
DATA_DIR = Path("/data/ax/ax_novas/data")
MILVUS_DIR = DATA_DIR / "milvus"
CHROMA_DIR = DATA_DIR / "chroma"
FAISS_DIR = DATA_DIR / "faiss"

# Vector dimensions (OpenAI embedding size)
VECTOR_DIM = 1536

class VectorStoreManager:
    """Manages the tiered vector store setup."""

    def __init__(self):
        self.milvus_collection: Optional[Collection] = None
        self.chroma_client: Optional[ChromaClient] = None
        self.faiss_index: Optional[faiss.Index] = None

    async def setup_directories(self) -> None:
        """Create necessary directories."""
        for directory in [MILVUS_DIR, CHROMA_DIR, FAISS_DIR]:
            directory.mkdir(parents=True, exist_ok=True)
            console.print(f"Created directory: {directory}")

    async def setup_milvus(self) -> None:
        """Setup Milvus with high-performance configuration."""
        # Connect to Milvus
        connections.connect(
            host='localhost',
            port=19530
        )

        # Define collection schema
        fields = [
            FieldSchema(
                name="id",
                dtype=DataType.INT64,
                is_primary=True,
                auto_id=True
            ),
            FieldSchema(
                name="embedding",
                dtype=DataType.FLOAT_VECTOR,
                dim=VECTOR_DIM
            ),
            FieldSchema(
                name="metadata",
                dtype=DataType.JSON
            )
        ]

        schema = CollectionSchema(
            fields=fields,
            description="NOVA memory vectors"
        )

        # Create collection with optimized settings
        self.milvus_collection = Collection(
            name="nova_memories",
            schema=schema,
            properties={
                "collection.ttl.seconds": 0,  # No TTL
                "segment.max_size": 4096,  # Larger segments for better performance
                "auto_compact": True,
                "compact.trigger": "0.3"  # Compact at 30% deleted
            }
        )

        # Create index with GPU support
        index_params = {
            "index_type": "IVF_SQ8",
            "metric_type": "IP",  # Inner product
            "params": {
                "nlist": 1024,
                "gpu_id": 0
            }
        }
        self.milvus_collection.create_index(
            field_name="embedding",
            index_params=index_params
        )

        # Load collection into memory
        self.milvus_collection.load(
            replica_number=1,
            using=["gpu0"]
        )

        console.print("[green]Milvus setup complete[/green]")

    async def setup_chroma(self) -> None:
        """Setup Chroma for working memory."""
        settings = ChromaSettings(
            persist_directory=str(CHROMA_DIR),
            anonymized_telemetry=False,
            allow_reset=False,
            is_persistent=True
        )

        self.chroma_client = ChromaClient(settings)
        
        # Create collection with optimized settings
        collection = self.chroma_client.create_collection(
            name="nova_working_memory",
            metadata={
                "hnsw:space": "ip",  # Inner product
                "hnsw:construction_ef": 200,
                "hnsw:search_ef": 100,
                "hnsw:max_elements": 1_000_000
            }
        )

        console.print("[green]Chroma setup complete[/green]")

    async def setup_faiss(self) -> None:
        """Setup FAISS with GPU support."""
        # Create high-performance index
        cpu_index = faiss.index_factory(
            VECTOR_DIM,
            "IVF1024,Flat",
            faiss.METRIC_INNER_PRODUCT
        )

        # Convert to GPU index
        res = faiss.StandardGpuResources()
        self.faiss_index = faiss.index_cpu_to_gpu(
            res,
            0,  # GPU id
            cpu_index
        )

        # Set search parameters
        self.faiss_index.nprobe = 64  # Number of clusters to search

        console.print("[green]FAISS setup complete[/green]")

    async def optimize_performance(self) -> None:
        """Apply performance optimizations."""
        # Milvus optimizations
        if self.milvus_collection:
            await self._optimize_milvus()

        # Chroma optimizations
        if self.chroma_client:
            await self._optimize_chroma()

        # FAISS optimizations
        if self.faiss_index:
            await self._optimize_faiss()

    async def _optimize_milvus(self) -> None:
        """Apply Milvus-specific optimizations."""
        # Set cache sizes
        utility.set_cache_size(
            cache_size="32GB",
            cache_type="cpu"
        )

        # Configure memory allocator
        utility.configure_memory_allocator(
            cache_size="64GB",
            prealloc_size="32GB"
        )

        console.print("[green]Milvus optimizations applied[/green]")

    async def _optimize_chroma(self) -> None:
        """Apply Chroma-specific optimizations."""
        # Configure persistence settings
        self.chroma_client.persist()

        console.print("[green]Chroma optimizations applied[/green]")

    async def _optimize_faiss(self) -> None:
        """Apply FAISS-specific optimizations."""
        # Save optimized index
        faiss.write_index(
            self.faiss_index,
            str(FAISS_DIR / "nova_vectors.index")
        )

        console.print("[green]FAISS optimizations applied[/green]")

    async def verify_setup(self) -> bool:
        """Verify the setup is working correctly."""
        success = True
        
        # Test vector operations
        test_vector = np.random.rand(VECTOR_DIM).astype('float32')
        
        try:
            # Test Milvus
            if self.milvus_collection:
                entities = [
                    [0],  # id
                    [test_vector.tolist()],  # embedding
                    [{"test": "data"}]  # metadata
                ]
                self.milvus_collection.insert(entities)
                
            # Test Chroma
            if self.chroma_client:
                collection = self.chroma_client.get_collection("nova_working_memory")
                collection.add(
                    embeddings=[test_vector.tolist()],
                    metadatas=[{"test": "data"}],
                    ids=["test_id"]
                )
                
            # Test FAISS
            if self.faiss_index:
                self.faiss_index.add(test_vector.reshape(1, -1))
                
        except Exception as e:
            console.print(f"[red]Verification failed: {str(e)}[/red]")
            success = False
            
        return success

async def main():
    """Main setup function."""
    with Progress(
        SpinnerColumn(),
        TextColumn("[progress.description]{task.description}"),
        transient=True
    ) as progress:
        
        # Initialize manager
        manager = VectorStoreManager()
        
        # Setup steps
        progress.add_task("Creating directories...", total=None)
        await manager.setup_directories()
        
        progress.add_task("Setting up Milvus...", total=None)
        await manager.setup_milvus()
        
        progress.add_task("Setting up Chroma...", total=None)
        await manager.setup_chroma()
        
        progress.add_task("Setting up FAISS...", total=None)
        await manager.setup_faiss()
        
        progress.add_task("Applying optimizations...", total=None)
        await manager.optimize_performance()
        
        progress.add_task("Verifying setup...", total=None)
        if await manager.verify_setup():
            console.print("\n[bold green]Vector store setup completed successfully![/bold green]")
        else:
            console.print("\n[bold red]Vector store setup encountered issues.[/bold red]")

if __name__ == "__main__":
    asyncio.run(main())</content>
