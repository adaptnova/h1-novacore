from datetime import datetime, timedelta
from typing import Any, Dict, List, Optional, Set, Tuple, Union

class VectorStoreAdapter:
    """Adapter for vector store backends."""

    def __init__(
        self,
        config: VectorStoreConfig,
        metrics_collector: MetricsCollector,
        tracing_manager: TracingManager
    ):
        self.config = config
        self.metrics = metrics_collector
        self.tracing = tracing_manager
        self._client = None

    async def initialize(self) -> None:
        """Initialize vector store connection."""
        with self.tracing.start_span("initialize_vector_store") as span:
            try:
                match self.config.provider:
                    case "milvus":
                        from pymilvus import connections
                        connections.connect(
                            alias="default",
                            host=self.config.host,
                            port=self.config.port
                        )
                    case "chroma":
                        from chromadb import Client
                        self._client = Client(
                            self.config.settings
                        )
                    case "faiss":
                        import faiss
                        self._client = faiss.read_index(
                            self.config.index_path
                        )
                        
                await self.metrics.record_metric(
                    "vector_store_initialized",
                    1,
                    {"provider": self.config.provider}
                )
                
            except Exception as e:
                span.record_exception(e)
                await self.metrics.record_metric(
                    "vector_store_initialization_failed",
                    1,
                    {"provider": self.config.provider}
                )
                raise

    async def store(
        self,
        vectors: List[List[float]],
        metadata: List[Dict[str, Any]]
    ) -> None:
        """Store vectors in the vector store."""
        with self.tracing.start_span("store_vectors") as span:
            try:
                match self.config.provider:
                    case "milvus":
                        from pymilvus import Collection
                        collection = Collection("nova_memories")
                        collection.insert([
                            vectors,
                            metadata
                        ])
                    case "chroma":
                        self._client.add(
                            embeddings=vectors,
                            metadatas=metadata
                        )
                    case "faiss":
                        self._client.add(
                            np.array(vectors).astype('float32')
                        )
                        
                await self.metrics.record_metric(
                    "vectors_stored",
                    len(vectors),
                    {"provider": self.config.provider}
                )
                
            except Exception as e:
                span.record_exception(e)
                await self.metrics.record_metric(
                    "vector_store_failed",
                    1,
                    {"provider": self.config.provider}
                )
                raise

    async def search(
        self,
        query_vector: List[float],
        limit: int = 10
    ) -> List[Tuple[List[float], float, Dict[str, Any]]]:
        """Search for similar vectors."""
        with self.tracing.start_span("search_vectors") as span:
            try:
                start_time = datetime.utcnow()
                
                match self.config.provider:
                    case "milvus":
                        from pymilvus import Collection
                        collection = Collection("nova_memories")
                        results = collection.search(
                            data=[query_vector],
                            anns_field="embedding",
                            param={"nprobe": 64},
                            limit=limit
                        )
                        matches = [
                            (r.embedding, r.distance, r.metadata)
                            for r in results[0]
                        ]
                    case "chroma":
                        results = self._client.query(
                            query_embeddings=[query_vector],
                            n_results=limit
                        )
                        matches = [
                            (r.embedding, r.distance, r.metadata)
                            for r in results
                        ]
                    case "faiss":
                        D, I = self._client.search(
                            np.array([query_vector]).astype('float32'),
                            limit
                        )
                        matches = [
                            (self._client.reconstruct(int(i)), d, {})
                            for d, i in zip(D[0], I[0])
                        ]
                        
                duration = (datetime.utcnow() - start_time).total_seconds()
                await self.metrics.record_metric(
                    "search_latency",
                    duration,
                    {"provider": self.config.provider}
                )
                
                return matches
                
            except Exception as e:
                span.record_exception(e)
                await self.metrics.record_metric(
                    "search_failed",
                    1,
                    {"provider": self.config.provider}
                )
                raise


class MemoryService:
    """Service for managing system memory and context."""

    def __init__(
        self,
        memory_store: MemoryStore,
        metrics_collector: MetricsCollector,
        tracing_manager: TracingManager,
        vector_store_config: VectorStoreConfig
    ):
        self.memory_store = memory_store
        self.metrics_collector = metrics_collector
        self.tracing_manager = tracing_manager
        self.vector_store = VectorStoreAdapter(
            vector_store_config,
            metrics_collector,
            tracing_manager
        )
        self.embeddings = OpenAIEmbeddings()
        self.semantic_cache: Dict[str, Tuple[List[float], datetime]] = {}
        self.context_cache: Dict[UUID, Context] = {}

    async def initialize(self) -> None:
        """Initialize memory service."""
        with self.tracing_manager.start_span("initialize_memory_service") as span:
            try:
                # Initialize vector store
                await self.vector_store.initialize()
                
                # Initialize memory store
                await self.memory_store.initialize()
                
            except Exception as e:
                span.record_exception(e)
                raise

    async def store_memory(
        self,
        content: Any,
        memory_type: str,
        context: Optional[Dict[str, Any]] = None,
        metadata: Optional[Dict[str, Any]] = None
    ) -> Memory:
        """Store a new memory."""
        with self.tracing_manager.start_span("store_memory") as span:
            try:
                # Generate embedding if content is text
                embedding = None
                if isinstance(content, str):
                    embedding = await self._generate_embedding(content)

                # Create memory
                memory = Memory(
                    type=memory_type,
                    content=content,
                    context=context or {},
                    embedding=embedding.tolist() if embedding is not None else None,
                    metadata=metadata or {}
                )

                # Store memory
                await self.memory_store.store(memory)

                # Store in vector store if embedding exists
                if embedding is not None:
                    await self.vector_store.store(
                        [embedding.tolist()],
                        [{"memory_id": str(memory.id)}]
                    )

                # Update cache
                if embedding is not None:
                    self.semantic_cache[content] = (embedding, datetime.utcnow())

                # Record metrics
                await self.metrics_collector.record_metric(
                    "memories_stored_total",
                    1,
                    {"type": memory_type}
                )

                return memory

            except Exception as e:
                span.record_exception(e)
                raise

    async def retrieve_memories(
        self,
        query: str,
        memory_type: Optional[str] = None,
        limit: int = 5,
        threshold: float = 0.7,
        context: Optional[Dict[str, Any]] = None
    ) -> List[Memory]:
        """Retrieve relevant memories."""
        with self.tracing_manager.start_span("retrieve_memories") as span:
            try:
                # Check cache first
                cached_result = self._check_cache(query)
                if cached_result is not None:
                    return cached_result

                # Generate query embedding
                query_embedding = await self._generate_embedding(query)

                # Search vector store

                    query_embedding.tolist(),

                            memories.append(memory)

class FAISSConfig(BaseModel):

    index_path: Path = Field(default=Path("/data/ax/ax_novas/data/faiss/primary.index"))
    backup_path: Path = Field(default=Path("/data/ax/ax_novas/data/faiss/backup.index"))
    dimension: int = Field(default=1536)  # OpenAI embedding size

    cache_size_gb: int = Field(default=32)  # GPU cache size

class Memory(Resource):
    """Base memory model."""
    
    content: str
    context: Context
    source: str
    memory_type: str
    importance: float = Field(default=0.0)
    last_accessed: datetime = Field(default_factory=datetime.utcnow)
    access_count: int = Field(default=0)
    tags: Set[str] = Field(default_factory=set)
    metadata: Dict[str, Any] = Field(default_factory=dict)
    embedding: Optional[List[float]] = None

    def update_access(self) -> None:
        """Update memory access information."""
        self.last_accessed = datetime.utcnow()
        self.access_count += 1


class EpisodicMemory(Memory):
    """Episodic memory model."""
    
    episode_time: datetime
    location: Optional[str] = None
    participants: List[str] = Field(default_factory=list)
    emotions: Dict[str, float] = Field(default_factory=dict)
    outcomes: List[str] = Field(default_factory=list)


class SemanticMemory(Memory):
    """Semantic memory model."""

    relationships: Dict[str, List[str]] = Field(default_factory=dict)
    confidence: float = Field(default=1.0)
    sources: List[str] = Field(default_factory=list)

class WorkingMemory(BaseModel):
    """Working memory model."""
    
    capacity: int = Field(default=7)
    items: List[Memory] = Field(default_factory=list)
    priority_queue: List[UUID] = Field(default_factory=list)
    context_window: Context = Field(default_factory=Context)

        """Add item to working memory."""
        if len(self.items) >= self.capacity:
            self.items.pop(0)
        self.items.append(item)

        """Get current context."""
        return self.context_window

    def update_context(self, context: Context) -> None:
        """Update context window."""
        self.context_window = context


class FAISSMemoryStore:
    """FAISS-based memory store."""
    
    def __init__(self, config: FAISSConfig):
        self.config = config
        self.index = None
        self.gpu_resources = None
        self.metadata: Dict[int, Dict[str, Any]] = {}
        self.next_id: int = 0

    async def initialize(self) -> None:
        """Initialize FAISS index."""
        # Create CPU index
        cpu_index = faiss.index_factory(
            self.config.dimension,
            self.config.index_type,
            faiss.METRIC_INNER_PRODUCT
            if self.config.metric_type == "inner_product"
            else faiss.METRIC_L2
        )

        # Convert to GPU index
        self.gpu_resources = faiss.StandardGpuResources()
        self.gpu_resources.setTempMemory(self.config.cache_size_gb * 1024 * 1024 * 1024)
        
        self.index = faiss.index_cpu_to_gpu(
            self.gpu_resources,
            self.config.gpu_id,
            cpu_index
        )

        # Set search parameters
        self.index.nprobe = self.config.nprobe

        # Load existing index if available
        if self.config.index_path.exists():
            self.index = faiss.read_index(str(self.config.index_path))
            self.next_id = self.index.ntotal

    async def store(
        self,
        memories: List[Memory],
        batch: bool = True
    ) -> List[int]:
        """Store memories in FAISS."""
        vectors = []
        ids = []
        
        for memory in memories:
            if memory.embedding:
                vector = np.array(memory.embedding).astype('float32')
                vectors.append(vector)
                
                memory_id = self.next_id
                self.next_id += 1
                ids.append(memory_id)
                
                # Store metadata
                self.metadata[memory_id] = {
                    "memory_id": str(memory.id),
                    "type": memory.memory_type,
                    "importance": memory.importance,
                    "tags": list(memory.tags)
                }

        if vectors:
            vectors = np.vstack(vectors)
            if batch and len(vectors) < self.config.batch_size:
                # Accumulate for batch insert
                return ids
            
            self.index.add(vectors)
            # Save index periodically
            if self.next_id % 10000 == 0:
                await self.save()
                
        return ids

    async def search(
        self,
        query_vector: np.ndarray,
        k: int = 10,
        threshold: float = 0.7
    ) -> List[Tuple[int, float, Dict[str, Any]]]:
        """Search for similar vectors."""
        query_vector = query_vector.reshape(1, -1).astype('float32')
        distances, indices = self.index.search(query_vector, k)
        
        results = []
        for distance, idx in zip(distances[0], indices[0]):
            if idx != -1 and (
                self.config.metric_type == "inner_product" and distance >= threshold
                or self.config.metric_type == "l2" and distance <= threshold
            ):
                results.append((
                    int(idx),
                    float(distance),
                    self.metadata.get(int(idx), {})
                ))
                
        return results

    async def save(self) -> None:
        """Save index and metadata."""
        # Save main index
        faiss.write_index(
            faiss.index_gpu_to_cpu(self.index),
            str(self.config.index_path)
        )
        
        # Save backup
        faiss.write_index(
            faiss.index_gpu_to_cpu(self.index),
            str(self.config.backup_path)
        )

    async def load(self) -> None:
        """Load index from disk."""
        if self.config.index_path.exists():
            cpu_index = faiss.read_index(str(self.config.index_path))
            self.index = faiss.index_cpu_to_gpu(
                self.gpu_resources,
                self.config.gpu_id,
                cpu_index
            )
            self.next_id = self.index.ntotal

    async def cleanup(self) -> None:
        """Clean up resources."""
        if self.index:
            await self.save()
        if self.gpu_resources:
            del self.gpu_resources


class MemoryStore(BaseModel):
    """Memory storage system."""
    
    episodic: Dict[UUID, EpisodicMemory] = Field(default_factory=dict)
    semantic: Dict[UUID, SemanticMemory] = Field(default_factory=dict)
    procedural: Dict[UUID, ProceduralMemory] = Field(default_factory=dict)
    working: WorkingMemory = Field(default_factory=WorkingMemory)
    vector_store: FAISSMemoryStore
    metadata: Metadata = Field(default_factory=Metadata)

    class Config:
        arbitrary_types_allowed = True

    async def initialize(self) -> None:
        """Initialize memory store."""
        await self.vector_store.initialize()

    async def store(self, memory: Memory) -> None:
        """Store a memory."""
        # Store in appropriate memory store
        if isinstance(memory, EpisodicMemory):
            self.episodic[memory.id] = memory
        elif isinstance(memory, SemanticMemory):
            self.semantic[memory.id] = memory
        elif isinstance(memory, ProceduralMemory):
            self.procedural[memory.id] = memory

        # Store in vector store if embedding exists
        if memory.embedding:
            await self.vector_store.store([memory])

    async def get(self, memory_id: UUID) -> Optional[Memory]:
        """Retrieve a memory by ID."""
        memory = (
            self.episodic.get(memory_id) or
            self.semantic.get(memory_id) or
            self.procedural.get(memory_id)
        )
        
        if memory:
            memory.update_access()
            
        return memory

    async def search(
        self,
        query_vector: np.ndarray,
        memory_type: Optional[str] = None,
        limit: int = 10,
        threshold: float = 0.7

            query_vector,
            k=limit * 2,  # Get extra results for filtering

        for _, score, metadata in results:

                if memory := await self.get(UUID(memory_id)):
                    if not memory_type or memory.memory_type == memory_type:
                        memories.append(memory)
                        
        return memories[:limit]
