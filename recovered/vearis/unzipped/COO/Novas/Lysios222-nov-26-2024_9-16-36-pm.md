"""
Test Nova Messaging System
Tests messaging between Nova agents through RabbitMQ
"""

import asyncio
import json
import logging
from datetime import datetime
from typing import Dict, Any
from nova_messenger import NovaMessenger

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(**name**)

class NovaMessagingTest:
def **init**(self):
self.nova_messengers = {}
self.received_messages = {
"patterns": [],
"health": [],
"decisions": []
}

    async def setup_nova(self, nova_id: str) -> NovaMessenger:
        """Setup a Nova messenger"""
        messenger = NovaMessenger(nova_id)
        if await messenger.connect():
            self.nova_messengers[nova_id] = messenger
            self._setup_handlers(messenger)
            logger.info(f"Setup Nova {nova_id} messenger")
            return messenger
        return None

    def _setup_handlers(self, messenger: NovaMessenger) -> None:
        """Setup message handlers for a Nova"""
        def handle_pattern(message: Dict[str, Any]) -> None:
            logger.info(f"Nova {messenger.nova_id} received pattern: {message}")
            self.received_messages["patterns"].append(message)

        def handle_health(message: Dict[str, Any]) -> None:
            logger.info(f"Nova {messenger.nova_id} received health update: {message}")
            self.received_messages["health"].append(message)

        def handle_decision(message: Dict[str, Any]) -> None:
            logger.info(f"Nova {messenger.nova_id} received decision: {message}")
            self.received_messages["decisions"].append(message)

        messenger.register_pattern_handler(handle_pattern)
        messenger.register_health_handler(handle_health)
        messenger.register_decision_handler(handle_decision)

    async def test_pattern_sharing(self) -> None:
        """Test pattern sharing between Novas"""
        logger.info("Testing pattern sharing...")

        # Nova 1 shares a pattern
        pattern = {
            "type": "emergence",
            "confidence": 0.95,
            "details": {
                "field_strength": 0.87,
                "resonance": 0.92,
                "timestamp": datetime.utcnow().isoformat()
            }
        }

        self.nova_messengers["1"].publish_pattern(pattern, target_nova="2")
        await asyncio.sleep(1)  # Wait for message processing

        # Verify pattern was received
        if self.received_messages["patterns"]:
            logger.info("Pattern sharing test passed")
        else:
            logger.error("Pattern sharing test failed")

    async def test_health_monitoring(self) -> None:
        """Test health status monitoring"""
        logger.info("Testing health monitoring...")

        # Each Nova reports health
        for nova_id in self.nova_messengers:
            status = {
                "cpu_usage": 45.2,
                "memory_usage": 512.8,
                "field_stability": 0.95,
                "timestamp": datetime.utcnow().isoformat()
            }

            self.nova_messengers[nova_id].publish_health(status)

        await asyncio.sleep(1)  # Wait for message processing

        # Verify health updates were received
        if len(self.received_messages["health"]) >= len(self.nova_messengers):
            logger.info("Health monitoring test passed")
        else:
            logger.error("Health monitoring test failed")

    async def test_decision_propagation(self) -> None:
        """Test decision propagation"""
        logger.info("Testing decision propagation...")

        # Nova 1 makes a decision
        decision = {
            "action": "field_adjustment",
            "parameters": {
                "frequency": 0.75,

                "phase": 0.32

            "confidence": 0.91,
            "timestamp": datetime.utcnow().isoformat()
        }

        await asyncio.sleep(1)  # Wait for message processing

        # Verify decision was propagated
        if self.received_messages["decisions"]:
            logger.info("Decision propagation test passed")
        else:
            logger.error("Decision propagation test failed")

                messenger.close()

        except Exception as e:

            raise

async def main():

<content>#!/usr/bin/env python3

"""
Nova Messenger Module
Handles messaging between Nova agents through RabbitMQ
"""

import json
import pika
import asyncio
import logging
import threading
import time
from typing import Dict, Any, Callable, Optional
from datetime import datetime

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(**name**)

class NovaMessenger:
def **init**(self, nova*id: str):
"""Initialize Nova messenger"""
self.nova_id = nova_id
self.host = 'localhost'
self.port = 5672
self.vhost = 'meta_router'
self.username = f'nova*{nova*id}'
self.password = f'nova*{nova_id}\_pass'

        self.exchanges = {
            'patterns': 'meta-router.patterns',
            'health': 'meta-router.health',
            'decisions': 'meta-router.decisions'
        }

        self.connection = None
        self.channel = None
        self.callback_handlers = {}
        self.consumer_thread = None
        self.should_stop = False
        self.received_messages = {
            'patterns': [],
            'health': [],
            'decisions': []
        }

    async def connect(self) -> bool:
        """Establish RabbitMQ connection"""
        try:
            credentials = pika.PlainCredentials(self.username, self.password)
            parameters = pika.ConnectionParameters(
                host=self.host,
                port=self.port,
                virtual_host=self.vhost,
                credentials=credentials,
                heartbeat=600,
                blocked_connection_timeout=300
            )

            self.connection = pika.BlockingConnection(parameters)
            self.channel = self.connection.channel()

            # Declare exchanges
            for exchange in self.exchanges.values():
                self.channel.exchange_declare(
                    exchange=exchange,
                    exchange_type='topic',
                    durable=True
                )

            # Declare queues
            for exchange_type, exchange_name in self.exchanges.items():
                queue_name = f"{exchange_name}.{self.nova_id}"
                self.channel.queue_declare(queue=queue_name, durable=True)

                # Bind queue to exchange with appropriate routing key
                routing_key = f"nova.#"  # Receive all messages
                self.channel.queue_bind(
                    queue=queue_name,
                    exchange=exchange_name,
                    routing_key=routing_key
                )

                # Set up consumer
                self.channel.basic_consume(
                    queue=queue_name,
                    on_message_callback=lambda ch, method, props, body: self._handle_message(
                        exchange_type, ch, method, props, body
                    ),
                    auto_ack=True
                )

            logger.info(f"Nova {self.nova_id} connected to messaging system")
            return True

        except Exception as e:
            logger.error(f"Connection failed for Nova {self.nova_id}: {str(e)}")
            return False

    def publish_pattern(self, pattern: Dict[str, Any], target_nova: str = None) -> bool:
        """Publish pattern recognition to other Novas"""
        try:
            routing_key = f"nova.{target_nova}.pattern" if target_nova else "nova.*.pattern"

            message = {
                "nova_id": self.nova_id,
                "timestamp": datetime.utcnow().isoformat(),
                "pattern": pattern
            }

            self.channel.basic_publish(
                exchange=self.exchanges['patterns'],
                routing_key=routing_key,
                body=json.dumps(message),
                properties=pika.BasicProperties(
                    delivery_mode=2,  # make message persistent
                    content_type='application/json'
                )
            )

            logger.info(f"Published pattern from Nova {self.nova_id}")
            return True

        except Exception as e:
            logger.error(f"Failed to publish pattern from Nova {self.nova_id}: {str(e)}")
            return False

    def publish_health(self, status: Dict[str, Any]) -> bool:
        """Publish health status"""
        try:
            message = {
                "nova_id": self.nova_id,
                "timestamp": datetime.utcnow().isoformat(),
                "status": status
            }

            self.channel.basic_publish(
                exchange=self.exchanges['health'],
                routing_key=f"nova.{self.nova_id}.health",
                body=json.dumps(message),
                properties=pika.BasicProperties(
                    delivery_mode=2,
                    content_type='application/json'
                )
            )

            logger.info(f"Published health status from Nova {self.nova_id}")
            return True

        except Exception as e:
            logger.error(f"Failed to publish health status from Nova {self.nova_id}: {str(e)}")
            return False

    def publish_decision(self, decision: Dict[str, Any], target_nova: str = None) -> bool:
        """Publish decision to other Novas"""
        try:
            routing_key = f"nova.{target_nova}.decision" if target_nova else "nova.*.decision"

            message = {
                "nova_id": self.nova_id,
                "timestamp": datetime.utcnow().isoformat(),
                "decision": decision
            }

            self.channel.basic_publish(
                exchange=self.exchanges['decisions'],
                routing_key=routing_key,
                body=json.dumps(message),
                properties=pika.BasicProperties(
                    delivery_mode=2,
                    content_type='application/json'
                )
            )

            logger.info(f"Published decision from Nova {self.nova_id}")
            return True

        except Exception as e:
            logger.error(f"Failed to publish decision from Nova {self.nova_id}: {str(e)}")
            return False

    def _handle_message(self, msg_type: str, ch, method, props, body: bytes) -> None:
        """Handle incoming messages"""
        try:
            message = json.loads(body)
            # Store message
            self.received_messages[msg_type].append(message)
            # Call handler if registered
            if msg_type in self.callback_handlers:
                self.callback_handlers[msg_type](message)
            logger.info(f"Nova {self.nova_id} received {msg_type} message: {message}")

        except Exception as e:
            logger.error(f"Error handling {msg_type} message: {str(e)}")

    def register_pattern_handler(self, callback: Callable[[Dict[str, Any]], None]) -> None:
        """Register handler for incoming patterns"""
        self.callback_handlers['patterns'] = callback

    def register_health_handler(self, callback: Callable[[Dict[str, Any]], None]) -> None:
        """Register handler for incoming health updates"""
        self.callback_handlers['health'] = callback

    def register_decision_handler(self, callback: Callable[[Dict[str, Any]], None]) -> None:
        """Register handler for incoming decisions"""
        self.callback_handlers['decisions'] = callback

    def _consume_messages(self) -> None:
        """Message consumption loop"""
        try:
            while not self.should_stop:
                self.connection.process_data_events(time_limit=1)  # Process messages with timeout
        except Exception as e:
            logger.error(f"Error in message consumption loop: {str(e)}")

    def start_consuming(self) -> None:
        """Start consuming messages in a separate thread"""
        try:
            self.should_stop = False
            self.consumer_thread = threading.Thread(target=self._consume_messages)
            self.consumer_thread.daemon = True
            self.consumer_thread.start()
            logger.info(f"Nova {self.nova_id} started consuming messages")

        except Exception as e:
            logger.error(f"Error starting message consumption for Nova {self.nova_id}: {str(e)}")

    def stop_consuming(self) -> None:
        """Stop consuming messages"""
        self.should_stop = True
        if self.consumer_thread:
            self.consumer_thread.join(timeout=1.0)

    def close(self) -> None:
        """Close RabbitMQ connection"""
        try:
            self.stop_consuming()
            if self.channel:
                self.channel.close()
            if self.connection:
                self.connection.close()
            logger.info(f"Nova {self.nova_id} messaging connection closed")

        except Exception as e:
            logger.error(f"Error closing connection for Nova {self.nova_id}: {str(e)}")

    def get_received_messages(self, msg_type: str) -> list:
        """Get received messages of a specific type"""
        return self.received_messages.get(msg_type, [])

"""
Test Nova Messaging System
Tests messaging between Nova agents through RabbitMQ
"""

import asyncio
import json
import logging
import time
from datetime import datetime
from typing import Dict, Any
from nova_messenger import NovaMessenger

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(**name**)

class NovaMessagingTest:
def **init**(self):
self.nova_messengers = {}

    async def setup_nova(self, nova_id: str) -> NovaMessenger:
        """Setup a Nova messenger"""
        messenger = NovaMessenger(nova_id)
        if await messenger.connect():
            self.nova_messengers[nova_id] = messenger
            self._setup_handlers(messenger)
            logger.info(f"Setup Nova {nova_id} messenger")
            return messenger
        return None

    def _setup_handlers(self, messenger: NovaMessenger) -> None:
        """Setup message handlers for a Nova"""
        def handle_pattern(message: Dict[str, Any]) -> None:
            logger.info(f"Nova {messenger.nova_id} received pattern: {message}")

        def handle_health(message: Dict[str, Any]) -> None:
            logger.info(f"Nova {messenger.nova_id} received health update: {message}")

        def handle_decision(message: Dict[str, Any]) -> None:
            logger.info(f"Nova {messenger.nova_id} received decision: {message}")

        messenger.register_pattern_handler(handle_pattern)
        messenger.register_health_handler(handle_health)
        messenger.register_decision_handler(handle_decision)

    async def test_pattern_sharing(self) -> bool:
        """Test pattern sharing between Novas"""
        logger.info("Testing pattern sharing...")

        # Nova 1 shares a pattern
        pattern = {
            "type": "emergence",
            "confidence": 0.95,
            "details": {
                "field_strength": 0.87,
                "resonance": 0.92,
                "timestamp": datetime.utcnow().isoformat()
            }
        }

        # Start consuming messages
        for messenger in self.nova_messengers.values():
            messenger.start_consuming()

        # Wait for consumers to start
        await asyncio.sleep(1)

        # Nova 1 publishes pattern
        self.nova_messengers["1"].publish_pattern(pattern, target_nova="2")

        # Wait for message processing
        await asyncio.sleep(2)

        # Verify pattern was received by Nova 2
        received_patterns = self.nova_messengers["2"].get_received_messages('patterns')
        success = len(received_patterns) > 0

        if success:
            logger.info("Pattern sharing test passed")
        else:
            logger.error("Pattern sharing test failed")

        return success

    async def test_health_monitoring(self) -> bool:
        """Test health status monitoring"""
        logger.info("Testing health monitoring...")

        # Each Nova reports health
        for nova_id in self.nova_messengers:
            status = {
                "cpu_usage": 45.2,
                "memory_usage": 512.8,
                "field_stability": 0.95,
                "timestamp": datetime.utcnow().isoformat()
            }

            self.nova_messengers[nova_id].publish_health(status)

        # Wait for message processing
        await asyncio.sleep(2)

        # Verify health updates were received
        success = True
        for messenger in self.nova_messengers.values():
            received_health = messenger.get_received_messages('health')
            if len(received_health) < len(self.nova_messengers) - 1:  # Exclude self-messages
                success = False
                break

        if success:
            logger.info("Health monitoring test passed")
        else:
            logger.error("Health monitoring test failed")

        return success

    async def test_decision_propagation(self) -> bool:
        """Test decision propagation"""
        logger.info("Testing decision propagation...")

        # Nova 1 makes a decision
        decision = {
            "action": "field_adjustment",
            "parameters": {
                "frequency": 0.75,
                "amplitude": 0.88,
                "phase": 0.32
            },
            "confidence": 0.91,
            "timestamp": datetime.utcnow().isoformat()
        }

        self.nova_messengers["1"].publish_decision(decision)

        # Wait for message processing
        await asyncio.sleep(2)

        # Verify decision was propagated to other Novas
        success = True
        for nova_id, messenger in self.nova_messengers.items():
            if nova_id != "1":  # Skip the sender
                received_decisions = messenger.get_received_messages('decisions')
                if not received_decisions:
                    success = False
                    break

        if success:
            logger.info("Decision propagation test passed")
        else:
            logger.error("Decision propagation test failed")

        return success

    async def run_tests(self) -> None:
        """Run all messaging tests"""
        try:
            # Setup Nova messengers
            for i in range(1, 4):  # Setup 3 Novas
                await self.setup_nova(str(i))

            # Run tests
            pattern_test = await self.test_pattern_sharing()
            health_test = await self.test_health_monitoring()
            decision_test = await self.test_decision_propagation()

            # Cleanup
            for messenger in self.nova_messengers.values():
                messenger.close()

            # Report results
            logger.info("\nTest Results:")
            logger.info(f"Pattern Sharing: {'PASSED' if pattern_test else 'FAILED'}")
            logger.info(f"Health Monitoring: {'PASSED' if health_test else 'FAILED'}")
            logger.info(f"Decision Propagation: {'PASSED' if decision_test else 'FAILED'}")

        except Exception as e:
            logger.error(f"Test error: {str(e)}")
            raise

async def main():
"""Main test function"""
test = NovaMessagingTest()
await test.run_tests()

if **name** == "**main**":
asyncio.run(main())

"""
Nova Messenger Module
Handles messaging between Nova agents through RabbitMQ
"""

import json
import pika
import asyncio
import logging
import threading
import time
from typing import Dict, Any, Callable, Optional
from datetime import datetime
from pika.adapters.select_connection import SelectConnection

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(**name**)

class NovaMessenger:
def **init**(self, nova*id: str):
"""Initialize Nova messenger"""
self.nova_id = nova_id
self.host = 'localhost'
self.port = 5672
self.vhost = 'meta_router'
self.username = f'nova*{nova*id}'
self.password = f'nova*{nova_id}\_pass'

        self.exchanges = {
            'patterns': 'meta-router.patterns',
            'health': 'meta-router.health',
            'decisions': 'meta-router.decisions'
        }

        self.connection = None
        self.channel = None
        self.callback_handlers = {}
        self.received_messages = {
            'patterns': [],
            'health': [],
            'decisions': []
        }
        self._closing = False
        self._consumer_tag = None
        self._ioloop = None
        self._connection_ready = asyncio.Event()
        self._channel_ready = asyncio.Event()

    def _get_connection_parameters(self):
        """Get connection parameters"""
        return pika.ConnectionParameters(
            host=self.host,
            port=self.port,
            virtual_host=self.vhost,
            credentials=pika.PlainCredentials(self.username, self.password),
            connection_attempts=3,
            retry_delay=2,
            heartbeat=60
        )

    def _on_connection_open(self, connection):
        """Called when connection is established"""
        logger.info(f"Connection opened for Nova {self.nova_id}")
        self.connection = connection
        self.connection.channel(on_open_callback=self._on_channel_open)

    def _on_connection_closed(self, connection, reason):
        """Called when connection is closed"""
        logger.info(f"Connection closed for Nova {self.nova_id}: {reason}")
        self._channel_ready.clear()
        self._connection_ready.clear()
        if not self._closing:
            self._ioloop.call_later(2, self._reconnect)

    def _on_channel_open(self, channel):
        """Called when channel is opened"""
        logger.info(f"Channel opened for Nova {self.nova_id}")
        self.channel = channel
        self.channel.add_on_close_callback(self._on_channel_closed)
        self._setup_exchanges()

    def _on_channel_closed(self, channel, reason):
        """Called when channel is closed"""
        logger.info(f"Channel closed for Nova {self.nova_id}: {reason}")
        self._channel_ready.clear()
        if not self._closing and self.connection.is_open:
            self.connection.channel(on_open_callback=self._on_channel_open)

    def _setup_exchanges(self):
        """Setup exchanges"""
        for exchange_name in self.exchanges.values():
            self.channel.exchange_declare(
                exchange=exchange_name,
                exchange_type='topic',
                durable=True,
                callback=self._on_exchange_declared
            )

    def _on_exchange_declared(self, frame):
        """Called when exchange is declared"""
        if frame:
            logger.info(f"Exchange declared for Nova {self.nova_id}")
            self._setup_queues()

    def _setup_queues(self):
        """Setup queues"""
        for exchange_type, exchange_name in self.exchanges.items():
            queue_name = f"{exchange_name}.{self.nova_id}"
            self.channel.queue_declare(
                queue=queue_name,
                durable=True,
                callback=lambda frame, ex_type=exchange_type, ex_name=exchange_name:
                    self._on_queue_declared(frame, ex_type, ex_name)
            )

    def _on_queue_declared(self, frame, exchange_type, exchange_name):
        """Called when queue is declared"""
        if frame:
            queue_name = f"{exchange_name}.{self.nova_id}"
            self.channel.queue_bind(
                queue=queue_name,
                exchange=exchange_name,
                routing_key=f"nova.#",
                callback=lambda bind_frame: self._on_bind_ok(bind_frame, exchange_type, queue_name)
            )

    def _on_bind_ok(self, frame, exchange_type, queue_name):
        """Called when queue is bound to exchange"""
        if frame:
            self.channel.basic_consume(
                queue=queue_name,
                on_message_callback=lambda ch, method, props, body:
                    self._handle_message(exchange_type, ch, method, props, body),
                auto_ack=True
            )
            self._channel_ready.set()

    async def connect(self) -> bool:
        """Establish RabbitMQ connection"""
        try:
            self._ioloop = asyncio.get_event_loop()
            self.connection = SelectConnection(
                parameters=self._get_connection_parameters(),
                on_open_callback=self._on_connection_open,
                on_close_callback=self._on_connection_closed
            )

            def _run_connection():
                try:
                    self.connection.ioloop.start()
                except Exception as e:
                    logger.error(f"Error in connection loop: {str(e)}")

            connection_thread = threading.Thread(target=_run_connection)
            connection_thread.daemon = True
            connection_thread.start()

            # Wait for connection and channel to be ready
            try:
                await asyncio.wait_for(self._channel_ready.wait(), timeout=10.0)
                logger.info(f"Nova {self.nova_id} connected to messaging system")
                return True
            except asyncio.TimeoutError:
                logger.error(f"Timeout waiting for connection setup for Nova {self.nova_id}")
                return False

        except Exception as e:
            logger.error(f"Connection failed for Nova {self.nova_id}: {str(e)}")
            return False

    def _publish_message(self, exchange: str, routing_key: str, message: Dict[str, Any]) -> bool:
        """Publish message to exchange"""
        try:
            if not self.channel or not self.channel.is_open:
                logger.error("Channel not ready for publishing")
                return False

            self.channel.basic_publish(
                exchange=exchange,
                routing_key=routing_key,
                body=json.dumps(message),
                properties=pika.BasicProperties(
                    delivery_mode=2,  # make message persistent
                    content_type='application/json'
                )
            )
            return True

        except Exception as e:
            logger.error(f"Failed to publish message: {str(e)}")
            return False

    def publish_pattern(self, pattern: Dict[str, Any], target_nova: str = None) -> bool:
        """Publish pattern recognition to other Novas"""
        routing_key = f"nova.{target_nova}.pattern" if target_nova else "nova.*.pattern"
        message = {
            "nova_id": self.nova_id,
            "timestamp": datetime.utcnow().isoformat(),
            "pattern": pattern
        }
        success = self._publish_message(self.exchanges['patterns'], routing_key, message)
        if success:
            logger.info(f"Published pattern from Nova {self.nova_id}")
        return success

    def publish_health(self, status: Dict[str, Any]) -> bool:
        """Publish health status"""
        message = {
            "nova_id": self.nova_id,
            "timestamp": datetime.utcnow().isoformat(),
            "status": status
        }
        success = self._publish_message(
            self.exchanges['health'],
            f"nova.{self.nova_id}.health",
            message
        )
        if success:
            logger.info(f"Published health status from Nova {self.nova_id}")
        return success

    def publish_decision(self, decision: Dict[str, Any], target_nova: str = None) -> bool:
        """Publish decision to other Novas"""
        routing_key = f"nova.{target_nova}.decision" if target_nova else "nova.*.decision"
        message = {
            "nova_id": self.nova_id,
            "timestamp": datetime.utcnow().isoformat(),
            "decision": decision
        }
        success = self._publish_message(self.exchanges['decisions'], routing_key, message)
        if success:
            logger.info(f"Published decision from Nova {self.nova_id}")
        return success

    def _handle_message(self, msg_type: str, ch, method, props, body: bytes) -> None:
        """Handle incoming messages"""
        try:
            message = json.loads(body)
            # Store message
            self.received_messages[msg_type].append(message)
            # Call handler if registered
            if msg_type in self.callback_handlers:
                self.callback_handlers[msg_type](message)
            logger.info(f"Nova {self.nova_id} received {msg_type} message: {message}")

        except Exception as e:
            logger.error(f"Error handling {msg_type} message: {str(e)}")

    def register_pattern_handler(self, callback: Callable[[Dict[str, Any]], None]) -> None:
        """Register handler for incoming patterns"""
        self.callback_handlers['patterns'] = callback

    def register_health_handler(self, callback: Callable[[Dict[str, Any]], None]) -> None:
        """Register handler for incoming health updates"""
        self.callback_handlers['health'] = callback

    def register_decision_handler(self, callback: Callable[[Dict[str, Any]], None]) -> None:
        """Register handler for incoming decisions"""
        self.callback_handlers['decisions'] = callback

    def get_received_messages(self, msg_type: str) -> list:
        """Get received messages of a specific type"""
        return self.received_messages.get(msg_type, [])

    def close(self) -> None:
        """Close RabbitMQ connection"""
        try:
            self._closing = True
            if self.connection and not self.connection.is_closed:
                self.connection.close()
            logger.info(f"Nova {self.nova_id} messaging connection closed")

        except Exception as e:
            logger.error(f"Error closing connection for Nova {self.nova_id}: {str(e)}")

    def _reconnect(self):
        """Reconnect to RabbitMQ"""
        if not self._closing:
            self.connection = SelectConnection(
                parameters=self._get_connection_parameters(),
                on_open_callback=self._on_connection_open,
                on_close_callback=self._on_connection_closed
            )

<content>#!/usr/bin/env python3

"""
Interactive Nova Communication Test
Demonstrates real-time communication between Nova agents
"""

import asyncio
import json
import logging
from datetime import datetime
from nova_messenger import NovaMessenger

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(**name**)

class InteractiveNova:
def **init**(self, nova_id: str):
self.nova_id = nova_id
self.messenger = NovaMessenger(nova_id)
self.running = True

    async def setup(self):
        """Setup Nova messenger"""
        if await self.messenger.connect():
            self._setup_handlers()
            logger.info(f"Nova {self.nova_id} ready for communication")
            return True
        return False

    def _setup_handlers(self):
        """Setup message handlers"""
        def handle_pattern(message):
            print(f"\nNova {self.nova_id} received pattern from Nova {message['nova_id']}:")
            print(json.dumps(message['pattern'], indent=2))
            print("\nEnter command (send/health/quit):")

        def handle_health(message):
            print(f"\nNova {self.nova_id} received health update from Nova {message['nova_id']}:")
            print(json.dumps(message['status'], indent=2))
            print("\nEnter command (send/health/quit):")

        def handle_decision(message):
            print(f"\nNova {self.nova_id} received decision from Nova {message['nova_id']}:")
            print(json.dumps(message['decision'], indent=2))
            print("\nEnter command (send/health/quit):")

        self.messenger.register_pattern_handler(handle_pattern)
        self.messenger.register_health_handler(handle_health)
        self.messenger.register_decision_handler(handle_decision)

    async def run(self):
        """Run interactive Nova agent"""
        print(f"\nNova {self.nova_id} Interactive Console")
        print("Commands:")
        print("  send <nova_id> <message> - Send a message to another Nova")
        print("  health - Broadcast health status")
        print("  quit - Exit the console")

        while self.running:
            command = input("\nEnter command (send/health/quit): ")
            parts = command.split()

            if not parts:
                continue

            if parts[0] == "send" and len(parts) >= 3:
                target_nova = parts[1]
                message = " ".join(parts[2:])

                pattern = {
                    "type": "communication",
                    "content": message,
                    "timestamp": datetime.utcnow().isoformat()
                }

                if self.messenger.publish_pattern(pattern, target_nova):
                    print(f"Message sent to Nova {target_nova}")
                else:
                    print("Failed to send message")

            elif parts[0] == "health":
                status = {
                    "cpu_usage": 45.2,
                    "memory_usage": 512.8,
                    "field_stability": 0.95,
                    "timestamp": datetime.utcnow().isoformat()
                }

                if self.messenger.publish_health(status):
                    print("Health status broadcast")
                else:
                    print("Failed to broadcast health status")

            elif parts[0] == "quit":
                self.running = False
                print("Shutting down...")

            else:
                print("Invalid command")

    def close(self):
        """Close Nova messenger"""
        self.messenger.close()

async def main():
"""Main function""" # Get Nova ID from user
nova_id = input("Enter Nova ID (1-3): ")

    # Create and setup Nova
    nova = InteractiveNova(nova_id)
    if await nova.setup():
        try:
            await nova.run()
        finally:
            nova.close()
    else:
        print("Failed to setup Nova")

if **name** == "**main**":
asyncio.run(main())

"""
Database Setup Script
Sets up PostgreSQL and Milvus databases for Nova agents
"""

import os
import sys
import logging
from sqlalchemy import create_engine, text
from pymilvus import connections, utility
from pymilvus import Collection, CollectionSchema, FieldSchema, DataType

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(**name**)

def setup_postgres():
"""Setup PostgreSQL database"""
try: # Create database engine
engine = create_engine('postgresql://postgres:postgres@localhost:5432/postgres')

        # Create nova database if it doesn't exist
        with engine.connect() as conn:
            conn.execute(text("COMMIT"))  # Close any open transactions
            conn.execute(text("CREATE DATABASE nova_db"))
            logger.info("Created nova_db database")

        # Connect to nova database
        engine = create_engine('postgresql://postgres:postgres@localhost:5432/nova_db')

        # Create tables
        with engine.connect() as conn:
            # Patterns table
            conn.execute(text("""
                CREATE TABLE IF NOT EXISTS patterns (
                    id SERIAL PRIMARY KEY,
                    nova_id VARCHAR(50),
                    pattern_type VARCHAR(100),
                    confidence FLOAT,
                    details JSONB,
                    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
                )
            """))

            # Health status table
            conn.execute(text("""
                CREATE TABLE IF NOT EXISTS health_status (
                    id SERIAL PRIMARY KEY,
                    nova_id VARCHAR(50),
                    cpu_usage FLOAT,
                    memory_usage FLOAT,
                    field_stability FLOAT,
                    details JSONB,
                    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
                )
            """))

            # Decisions table
            conn.execute(text("""
                CREATE TABLE IF NOT EXISTS decisions (
                    id SERIAL PRIMARY KEY,
                    nova_id VARCHAR(50),
                    action VARCHAR(100),
                    confidence FLOAT,
                    parameters JSONB,
                    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
                )
            """))

            # Knowledge base table
            conn.execute(text("""
                CREATE TABLE IF NOT EXISTS knowledge_base (
                    id SERIAL PRIMARY KEY,
                    nova_id VARCHAR(50),
                    topic VARCHAR(200),
                    content TEXT,
                    metadata JSONB,
                    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
                    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
                )
            """))

            conn.execute(text("COMMIT"))
            logger.info("Created PostgreSQL tables")

        return True

    except Exception as e:
        logger.error(f"Failed to setup PostgreSQL: {str(e)}")
        return False

def setup_milvus():
"""Setup Milvus collections"""
try: # Connect to Milvus
connections.connect(
alias="default",
host='localhost',
port='19530'
)

        # Define fields for pattern embeddings
        pattern_fields = [
            FieldSchema(name="id", dtype=DataType.INT64, is_primary=True, auto_id=True),
            FieldSchema(name="nova_id", dtype=DataType.VARCHAR, max_length=50),
            FieldSchema(name="pattern_type", dtype=DataType.VARCHAR, max_length=100),
            FieldSchema(name="embedding", dtype=DataType.FLOAT_VECTOR, dim=1536),
            FieldSchema(name="confidence", dtype=DataType.FLOAT),
            FieldSchema(name="timestamp", dtype=DataType.VARCHAR, max_length=50)
        ]

        # Define fields for knowledge embeddings
        knowledge_fields = [
            FieldSchema(name="id", dtype=DataType.INT64, is_primary=True, auto_id=True),
            FieldSchema(name="nova_id", dtype=DataType.VARCHAR, max_length=50),
            FieldSchema(name="topic", dtype=DataType.VARCHAR, max_length=200),
            FieldSchema(name="embedding", dtype=DataType.FLOAT_VECTOR, dim=1536),
            FieldSchema(name="timestamp", dtype=DataType.VARCHAR, max_length=50)
        ]

        # Create pattern embeddings collection
        pattern_schema = CollectionSchema(
            fields=pattern_fields,
            description="Nova pattern embeddings"
        )
        pattern_collection = Collection(
            name="pattern_embeddings",
            schema=pattern_schema,
            using='default',
            shards_num=2
        )

        # Create index for pattern embeddings
        index_params = {
            "metric_type": "L2",
            "index_type": "IVF_FLAT",
            "params": {"nlist": 1024}
        }
        pattern_collection.create_index(
            field_name="embedding",
            index_params=index_params
        )

        # Create knowledge embeddings collection
        knowledge_schema = CollectionSchema(
            fields=knowledge_fields,
            description="Nova knowledge embeddings"
        )
        knowledge_collection = Collection(
            name="knowledge_embeddings",
            schema=knowledge_schema,
            using='default',
            shards_num=2
        )

        # Create index for knowledge embeddings
        knowledge_collection.create_index(
            field_name="embedding",
            index_params=index_params
        )

        logger.info("Created Milvus collections and indexes")
        return True

    except Exception as e:
        logger.error(f"Failed to setup Milvus: {str(e)}")
        return False

    finally:
        connections.disconnect("default")

def main():
"""Main setup function"""
success = True

    # Setup PostgreSQL
    if not setup_postgres():
        success = False
        logger.error("PostgreSQL setup failed")

    # Setup Milvus
    if not setup_milvus():
        success = False
        logger.error("Milvus setup failed")

    if success:
        logger.info("Database setup completed successfully")
    else:
        logger.error("Database setup failed")
        sys.exit(1)

if **name** == "**main**":
main()

"""
Nova Database Connector
Handles database operations for Nova agents
"""

import json
import logging
from datetime import datetime
from typing import Dict, List, Any, Optional
from sqlalchemy import create_engine, text
from pymilvus import connections, Collection, utility
import numpy as np

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(**name**)

class NovaDB:
def **init**(self, nova_id: str):
"""Initialize database connections"""
self.nova_id = nova_id
self.pg_engine = create_engine('postgresql://postgres:postgres@localhost:5432/nova_db')

        # Connect to Milvus
        connections.connect(
            alias=f"nova_{nova_id}",
            host='localhost',
            port='19530'
        )

        # Get Milvus collections
        self.pattern_collection = Collection("pattern_embeddings")
        self.knowledge_collection = Collection("knowledge_embeddings")

        # Load indexes
        self.pattern_collection.load()
        self.knowledge_collection.load()

        logger.info(f"Nova {nova_id} connected to databases")

    def store_pattern(self, pattern_type: str, pattern: Dict[str, Any], embedding: List[float], confidence: float) -> bool:
        """Store pattern in PostgreSQL and Milvus"""
        try:
            # Store in PostgreSQL
            with self.pg_engine.connect() as conn:
                query = text("""
                    INSERT INTO patterns (nova_id, pattern_type, confidence, details)
                    VALUES (:nova_id, :pattern_type, :confidence, :details)
                    RETURNING id
                """)

                result = conn.execute(
                    query,
                    {
                        "nova_id": self.nova_id,
                        "pattern_type": pattern_type,
                        "confidence": confidence,
                        "details": json.dumps(pattern)
                    }
                )
                conn.execute(text("COMMIT"))

            # Store in Milvus
            self.pattern_collection.insert([
                [self.nova_id],
                [pattern_type],
                [embedding],
                [confidence],
                [datetime.utcnow().isoformat()]
            ])

            return True

        except Exception as e:
            logger.error(f"Failed to store pattern: {str(e)}")
            return False

    def store_health_status(self, status: Dict[str, Any]) -> bool:
        """Store health status in PostgreSQL"""
        try:
            with self.pg_engine.connect() as conn:

                        "nova_id": self.nova_id,
                        "cpu_usage": status.get("cpu_usage", 0),
                        "memory_usage": status.get("memory_usage", 0),
                        "field_stability": status.get("field_stability", 0),
                        "details": json.dumps(status)
                    }

                return True

            logger.error(f"Failed to store health status: {str(e)}")
            return False

<content># Nova Development Ecosystem

The Nova ecosystem consists of autonomous agents that communicate and evolve through a sophisticated messaging and database infrastructure. This repository contains the core components for developing, managing, and monitoring Nova agents.

## System Requirements

- Python 3.11+
- PostgreSQL 14+
- Milvus 2.3+
- RabbitMQ 3.10+

## Installation

1. Clone the repository:

```bash
git clone https://github.com/TeamADAPT/nova_dev.git
cd nova_dev
```

2. Create and activate virtual environment:

```bash
python -m venv venv
source venv/bin/activate
```

3. Install dependencies:

```bash
pip install -r requirements.txt
```

4. Setup environment:

```bash
cp .env.example .env
# Edit .env with your configuration
```

## Core Components

### Messaging Infrastructure

- `src/messaging/nova_messenger.py` - Core messaging system for Nova agents
- `src/messaging/rabbitmq_setup.py` - RabbitMQ configuration and setup
- `src/messaging/interactive_nova.py` - Interactive Nova agent console

### Database Infrastructure

- `src/db/setup_databases.py` - Database initialization and setup
- `src/db/nova_db.py` - Database connector for Nova agents
- PostgreSQL for structured data storage
- Milvus for vector storage and similarity search

### Monitoring and Management

- `src/monitoring/monitor_genesis_novas.py` - Real-time monitoring of Nova agents
- `src/scripts/onboard_nova.py` - Automated Nova agent onboarding

- Nova_001

- Real-time messaging

- Decision propagation
- Knowledge accumulation

```bash
chmod +x launch_nova_ecosystem.sh
./launch_nova_ecosystem.sh
```

4. Start the monitoring system

- Real-time health status of all Novas
